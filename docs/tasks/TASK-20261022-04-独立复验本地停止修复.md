## 可复现的隔离片段

使用项目venv执行以下Python片段；无真实进程启动/网络/生产库操作。原始输出missing、unknown均为expected_halt=True但actual=False，正常对照均False。

```python
import tempfile, json
from pathlib import Path
from datetime import datetime, timezone
from unittest.mock import Mock
from trademind.execution.auto_paper import Supervisor, result_policy

class Finishes:
    def poll(self): return None
    def wait(self, timeout=None): return 0
    def kill(self): pass

cases = [('missing', None), ('unknown', {
    'status': 'incomplete', 'model_pending': True,
    'shutdown': {'worker_active': True, 'unconsumed_result': True}}),
    ('classifiable', {'status': 'completed', 'model_pending': False,
    'shutdown': {'worker_active': False, 'unconsumed_result': False}})]
for name, report in cases:
    with tempfile.TemporaryDirectory() as folder:
        root = Path(folder)
        cal = Mock()
        cal.covers.return_value = True
        cal.is_session_open.return_value = True
        sup = Supervisor('unused', root, cal,
            clock=lambda: datetime(2026, 10, 22, 6, tzinfo=timezone.utc),
            account_id='audit', db_path=root/'absent.db')
        sup.child, sup.kind, sup.stopping = Finishes(), 'paper', True
        sup.report_path = root/'paper.json'
        if report is not None:
            sup.report_path.write_text(json.dumps(report), encoding='utf-8')
        expected = result_policy(report or {})[0]
        sup.shutdown()
        actual = json.loads(sup.path.read_text(encoding='utf-8'))['halted']
        print(name, 'expected_halt=', expected, 'actual=', actual)
```
